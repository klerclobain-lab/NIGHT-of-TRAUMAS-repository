"use client"

import { useEffect, useState } from "react"

const TARGET = new Date("2026-10-31T23:00:00+01:00").getTime()

function diff() {
  const d = Math.max(0, TARGET - Date.now())
  return {
    days: Math.floor(d / 86400000),
    hours: Math.floor((d / 3600000) % 24),
    minutes: Math.floor((d / 60000) % 60),
    seconds: Math.floor((d / 1000) % 60),
  }
}

export function Countdown() {
  const [time, setTime] = useState<ReturnType<typeof diff> | null>(null)

  useEffect(() => {
    setTime(diff())
    const id = setInterval(() => setTime(diff()), 1000)
    return () => clearInterval(id)
  }, [])

  const units: [string, number][] = time
    ? [
        ["GIORNI", time.days],
        ["ORE", time.hours],
        ["MIN", time.minutes],
        ["SEC", time.seconds],
      ]
    : [
        ["GIORNI", 0],
        ["ORE", 0],
        ["MIN", 0],
        ["SEC", 0],
      ]

  return (
    <div
      className="grid grid-cols-4 gap-2 sm:gap-3"
      role="timer"
      aria-label="Conto alla rovescia all'evento"
    >
      {units.map(([label, value]) => (
        <div
          key={label}
          className="flex flex-col items-center rounded-md border border-[#2a1215] bg-void/60 px-2 py-3 backdrop-blur-sm box-glow"
        >
          <span
            className="font-mono text-2xl font-bold tabular-nums text-neon text-glow sm:text-4xl"
            suppressHydrationWarning
          >
            {String(value).padStart(2, "0")}
          </span>
          <span className="mt-1 font-mono text-[9px] tracking-[0.3em] text-muted-foreground sm:text-[10px]">
            {label}
          </span>
        </div>
      ))}
    </div>
  )
}
