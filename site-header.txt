"use client"

import { useEffect, useState } from "react"

export function SiteHeader() {
  const [scrolled, setScrolled] = useState(false)

  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 24)
    onScroll()
    window.addEventListener("scroll", onScroll, { passive: true })
    return () => window.removeEventListener("scroll", onScroll)
  }, [])

  return (
    <header
      className={`fixed inset-x-0 top-0 z-50 transition-all duration-500 ${
        scrolled
          ? "border-b border-[#2a1215] bg-void/80 backdrop-blur-md"
          : "border-b border-transparent bg-transparent"
      }`}
    >
      <div className="mx-auto flex max-w-5xl items-center justify-between gap-3 px-4 py-3">
        <a href="#top" className="group flex items-center gap-2.5 leading-none">
          <img
            src="/logo.png"
            alt="NIGHT of TRAUMAS 3.0"
            className="h-9 w-auto drop-shadow-[0_0_12px_rgba(255,0,51,0.4)] sm:h-10"
          />
          <span className="hidden font-mono text-[10px] tracking-[0.4em] text-muted-foreground sm:inline">
            3.0 // UNDERGROUND
          </span>
        </a>

        <div className="flex items-center gap-2 sm:gap-3">
          <span className="hidden items-center gap-2 rounded-full border border-neon/40 px-3 py-1.5 font-mono text-[10px] tracking-widest text-neon box-glow sm:flex">
            <span className="relative flex h-2 w-2">
              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-neon opacity-75" />
              <span className="relative inline-flex h-2 w-2 rounded-full bg-neon" />
            </span>
            STATUS: PRE-SALE OPEN
          </span>

        </div>
      </div>
    </header>
  )
}
