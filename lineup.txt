function IconLink({
  href,
  label,
  children,
}: {
  href: string
  label: string

}) {
  return (
    <a
      href={href}
      target="_blank"
      rel="noopener noreferrer"
      aria-label={label}
      className="flex h-9 w-9 items-center justify-center rounded-full border border-[#2a1215] text-silver transition-colors hover:border-neon hover:text-neon hover:text-glow"
    >
      {children}
    </a>
  )
}
